# 1. 用數值方法來算單變數微分和積分
# =====================================================================
def diff_1d(f, x, h=1e-5):
    """單變數數值微分（中央差分法 Central Difference）"""
    return (f(x + h) - f(x - h)) / (2 * h)


def integral_1d(f, low, high, n=1000):
    """單變數數值積分（辛普森法則 Simpson's Rule）"""
    if n % 2 == 1:
        n += 1
    h = (high - low) / n
    x = np.linspace(low, high, n + 1)
    y = f(x)
    return (h / 3.0) * (
        y[0] + y[-1] + 4 * np.sum(y[1:-1:2]) + 2 * np.sum(y[2:-2:2])
    )

# 2. 用蒙地卡羅法來算單變數積分
# =====================================================================
def mc_integral_1d(f, low, high, samples=200000):
    """單變數蒙地卡羅積分"""
    x = np.random.uniform(low, high, samples)
    return (high - low) * np.mean(f(x))

# 3 & 5. 用數值方法來算 n 變數微分(梯度)和積分(超體積 - 黎曼積分法)
# =====================================================================
def gradient_nd(f, x, h=1e-5):
    """n 變數數值微分（梯度 Gradient）"""
    x = np.array(x, dtype=float)
    grad = np.zeros_like(x)
    for i in range(len(x)):
        x_plus = x.copy()
        x_minus = x.copy()
        x_plus[i] += h
        x_minus[i] -= h
        grad[i] = (f(x_plus) - f(x_minus)) / (2 * h)
    return grad


def integral(f, low, high, k=10):
    """
    黎曼積分法（中點法則 Midpoint Riemann Sum）計算 n 變數積分（超體積）
    參數:
        f: 目標函數，需支援輸入形狀為 (N, n) 的二維陣列並回傳長度 N 的一維陣列
        low: 各維度下界陣列，長度為 n
        high: 各維度上界陣列，長度為 n
        k: 每個維度切割的等分數
    """
    low = np.array(low, dtype=float)
    high = np.array(high, dtype=float)
    n = len(low)

    if len(high) != n:
        raise ValueError("low 與 high 的陣列長度必須相同！")

    # 防呆機制：避免高維度時網格點數 k^n 過大導致記憶體崩潰
    if k**n > 1e8:
        raise MemoryError(
            f"維度災難警告：n={n} 維時，即使每維只切 {k} 等分，網格點數高達 {k}^{n}，"
            f"超出硬體記憶體負荷，無法使用黎曼網格法計算！"
        )

    dx = (high - low) / k
    d_vol = np.prod(dx)  # 微小超體積元素 dV

    # 建立每個維度的中點座標
    axes = [
        np.linspace(low[i] + dx[i] / 2.0, high[i] - dx[i] / 2.0, k)
        for i in range(n)
    ]
    mesh = np.meshgrid(*axes, indexing="ij")
    points = np.stack([m.ravel() for m in mesh], axis=-1)

    values = f(points)
    return np.sum(values) * d_vol

# 4 & 6. 用蒙地卡羅法來算 n 變數積分(超體積)
# =====================================================================
def mc_integral(f, low, high, samples=500000):
    """n 變數蒙地卡羅積分（超體積）"""
    low = np.array(low, dtype=float)
    high = np.array(high, dtype=float)
    n = len(low)

    if len(high) != n:
        raise ValueError("low 與 high 的陣列長度必須相同！")

    hyper_volume = np.prod(high - low)
    points = np.random.uniform(low, high, size=(samples, n))
    values = f(points)
    return hyper_volume * np.mean(values)

if __name__ == "__main__":
    np.random.seed(42)  # 固定隨機種子以便重現結果

    print("=" * 60)
    print("【測試 1 & 2】單變數微分與積分 (數值方法 vs 蒙地卡羅法)")
    print("=" * 60)
    # 測試函數: f(x) = x^3 + 2x
    # 在 x=2 處導數理論值: 3*(2^2) + 2 = 14
    # 在 [0, 2] 區間積分理論值: [x^4/4 + x^2]_0^2 = 4 + 4 = 8
    f_1d = lambda x: x**3 + 2 * x
    print(f"函數 f(x) = x^3 + 2x")
    print(f"1. 數值微分 f'(2)          = {diff_1d(f_1d, 2.0):.6f} (理論值: 14.000000)")
    print(f"1. 數值積分 ∫f(x)dx [0,2]  = {integral_1d(f_1d, 0, 2):.6f} (理論值: 8.000000)")
    print(f"2. 蒙地卡羅積分 [0,2]      = {mc_integral_1d(f_1d, 0, 2):.6f} (理論值: 8.000000)\n")

    print("=" * 60)
    print("【測試 3 & 4】n 變數微分(梯度)與一般超體積積分")
    print("=" * 60)
    # 測試函數: f(x, y, z) = x^2 + y^2 + z^2
    # 在 (1, 2, 3) 梯度理論值: [2, 4, 6]
    # 在 [0,1] x [0,2] x [0,3] 超體積積分理論值:
    # ∫_0^3 ∫_0^2 ∫_0^1 (x^2+y^2+z^2) dx dy dz = 6*(1/3) + 3*(8/3) + 2*(27/3) = 2 + 8 + 18 = 28
    f_nd = lambda x: np.sum(x**2, axis=-1)
    grad_res = gradient_nd(f_nd, [1.0, 2.0, 3.0])
    print(f"函數 f(x) = x1^2 + x2^2 + ... + xn^2")
    print(f"3. 3變數梯度 ∇f([1,2,3])   = {grad_res} (理論值: [2. 4. 6.])")
    print(f"3. 3變數黎曼積分 [0,1]x[0,2]x[0,3]   = {integral(f_nd, [0,0,0], [1,2,3], k=30):.6f} (理論值: 28.000000)")
    print(f"4. 3變數蒙地卡羅 [0,1]x[0,2]x[0,3]   = {mc_integral(f_nd, [0,0,0], [1,2,3], samples=500000):.6f} (理論值: 28.000000)\n")

    print("=" * 60)
    print("【測試 5】黎曼積分法 integral(f, low, high) 多維度測試")
    print("題目: f(x) = x1^2 + ... + xn^2, low = [0]*n, high = [1]*n")
    print("=" * 60)
    dimensions = [1, 2, 5, 100]
    # 依據維度調整每軸切割數 k，避免低維不準、高維跑太久
    k_settings = {1: 1000, 2: 200, 5: 15, 100: 2}

    for n in dimensions:
        low = [0.0] * n
        high = [1.0] * n
        exact = n / 3.0  # 理論值: n * ∫_0^1 x^2 dx = n / 3
        k = k_settings[n]
        t_start = time.time()
        try:
            ans = integral(f_nd, low, high, k=k)
            elapsed = time.time() - t_start
            print(
                f"[{n:3d} 維] (k={k:4d}) | 黎曼積分 = {ans:10.6f} | "
                f"理論值 = {exact:10.6f} | 誤差 = {abs(ans-exact):.6f} | 耗時 = {elapsed:.4f}s"
            )
        except MemoryError as err:
            print(f"[{n:3d} 維] (k={k:4d}) | 計算失敗 -> {err}")

    print("\n" + "=" * 60)
    print("【測試 6】蒙地卡羅法 mc_integral(f, low, high) 多維度測試")
    print("題目: f(x) = x1^2 + ... + xn^2, low = [0]*n, high = [1]*n")
    print("=" * 60)
    for n in dimensions:
        low = [0.0] * n
        high = [1.0] * n
        exact = n / 3.0
        t_start = time.time()
        ans = mc_integral(f_nd, low, high, samples=500000)
        elapsed = time.time() - t_start
        print(
            f"[{n:3d} 維] (N=50萬) | 蒙地卡羅 = {ans:10.6f} | "
            f"理論值 = {exact:10.6f} | 誤差 = {abs(ans-exact):.6f} | 耗時 = {elapsed:.4f}s"
        )
